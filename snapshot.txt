package ru.example.prichastie.model

/** Когда читается шаг: до Причастия, само Причащение, после. */
enum class Phase { BEFORE, COMMUNION, AFTER }

/** Язык текста и записи. */
enum class Language { MODERN, CHURCH_SLAVONIC }

/** Способ прохождения шага. Фиксируется при начале шага и не меняется. */
enum class ReadingMode { TEXT, AUDIO }

/** Особое поведение шага. */
enum class StepKind {
    NORMAL,
    /** Тропарь и кондак зависят от литургии (см. LiturgyVariant). */
    LITURGY_DEPENDENT,
    /** Само Таинство: чтения нет, пользователь только подтверждает. */
    COMMUNION
}

/** Какая литургия совершалась. Нужна для тропаря и кондака. */
enum class LiturgyVariant(val shortName: String) {
    CHRYSOSTOM("Иоанн Златоуст"),
    BASIL("Василий Великий"),
    GREGORY("Григорий Двоеслов")
}

data class StepGroup(
    val id: String,
    val phase: Phase,
    val title: String,
    val subtitle: String,
    /** Необязательная группа (три канона): её можно отключить в настройках. */
    val optional: Boolean = false
)

data class Step(
    val id: String,
    val groupId: String,
    val phase: Phase,
    val title: String,
    val subtitle: String,
    /** Длительность записи в секундах (пока примерная). */
    val audioSeconds: Int,
    /** Ключ текста. У двух чтений Символа веры он общий: "creed". */
    val textKey: String = id,
    val kind: StepKind = StepKind.NORMAL
)

@Composable
fun PrichastieTheme(
    darkTheme: Boolean = isSystemInDarkTheme(),
    // Dynamic color is available on Android 12+
    dynamicColor: Boolean = true,
    content: @Composable () -> Unit
) {
    val colorScheme = when {
        dynamicColor && Build.VERSION.SDK_INT >= Build.VERSION_CODES.S -> {
            val context = LocalContext.current
            if (darkTheme) dynamicDarkColorScheme(context) else dynamicLightColorScheme(context)
        }

        darkTheme -> DarkColorScheme
        else -> LightColorScheme
    }

    MaterialTheme(
        colorScheme = colorScheme,
        typography = Typography,
        content = content
    )
}

package ru.example.prichastie

import android.app.Application
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import kotlinx.coroutines.flow.SharingStarted
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.map
import kotlinx.coroutines.flow.stateIn
import kotlinx.coroutines.launch
import ru.example.prichastie.model.Catalog
import ru.example.prichastie.model.Phase
import ru.example.prichastie.model.Step

data class StepUi(
    val step: Step,
    val number: Int?,        // null, если шаг не входит в выбранный режим
    val included: Boolean,
    val done: Boolean
)

data class UiState(
    val fullMode: Boolean = true,
    val started: Boolean = false,
    val steps: List<StepUi> = emptyList(),
    val progress: Float = 0f,
    val progressBefore: Float = 0f,
    val progressAfter: Float = 0f
) {
    /** Чекбокс «Полное прохождение» блокируется, когда подготовка началась. */
    val modeLocked: Boolean get() = started
}

class MainViewModel(app: Application) : AndroidViewModel(app) {

    private val repo = ProgressRepository(app)

    /** Группы, которые можно не читать (сейчас это «Три канона»). */
    private val optionalGroupIds: Set<String> =
        Catalog.groups.filter { it.optional }.map { it.id }.toSet()

    val ui: StateFlow<UiState> = repo.state
        .map { buildUi(it) }
        .stateIn(viewModelScope, SharingStarted.WhileSubscribed(5_000), UiState())

    private fun buildUi(s: SavedState): UiState {
        var counter = 0
        val steps = Catalog.steps.map { st ->
            val included = s.fullMode || st.groupId !in optionalGroupIds
            StepUi(
                step = st,
                number = if (included) ++counter else null,
                included = included,
                done = included && st.id in s.doneIds
            )
        }
        return UiState(
            fullMode = s.fullMode,
            started = s.started,
            steps = steps,
            progress = weighted(steps),
            progressBefore = weighted(steps.filter { it.step.phase == Phase.BEFORE }),
            progressAfter = weighted(steps.filter { it.step.phase == Phase.AFTER })
        )
    }

    private fun weighted(list: List<StepUi>): Float {
        val inc = list.filter { it.included }
        val total = inc.sumOf { it.step.audioSeconds.toLong() }
        if (total == 0L) return 0f
        val done = inc.filter { it.done }.sumOf { it.step.audioSeconds.toLong() }
        return done.toFloat() / total
    }

    fun setFullMode(value: Boolean) = viewModelScope.launch {
        if (!ui.value.started) repo.setFullMode(value)   // после старта режим менять нельзя
    }

    fun start() = viewModelScope.launch { repo.markStarted() }

    /** Для шагов с текстом, а также для «Готово» у Причастия и «Приидите, ядите…». */
    fun setDone(id: String, done: Boolean) = viewModelScope.launch { repo.setDone(id, done) }

    fun restart() = viewModelScope.launch { repo.reset() }
}

===== app\src\main\java\ru\example\prichastie\ProgressRepository.kt =====
package ru.example.prichastie

import android.content.Context
import androidx.datastore.preferences.core.*
import androidx.datastore.preferences.preferencesDataStore
import kotlinx.coroutines.flow.Flow
import kotlinx.coroutines.flow.map

private val Context.dataStore by preferencesDataStore(name = "progress")

data class SavedState(
    val fullMode: Boolean = true,
    val started: Boolean = false,
    val doneIds: Set<String> = emptySet()
)

class ProgressRepository(private val context: Context) {

    private object Keys {
        val FULL = booleanPreferencesKey("full_mode")
        val STARTED = booleanPreferencesKey("started")
        val DONE = stringSetPreferencesKey("done_ids")
    }

    val state: Flow<SavedState> = context.dataStore.data.map { p ->
        SavedState(
            fullMode = p[Keys.FULL] ?: true,
            started = p[Keys.STARTED] ?: false,
            doneIds = p[Keys.DONE] ?: emptySet()
        )
    }

    suspend fun setFullMode(value: Boolean) {
        context.dataStore.edit { it[Keys.FULL] = value }
    }

    suspend fun markStarted() {
        context.dataStore.edit { it[Keys.STARTED] = true }
    }

    suspend fun setDone(id: String, done: Boolean) {
        context.dataStore.edit { p ->
            val cur = p[Keys.DONE] ?: emptySet()
            p[Keys.DONE] = if (done) cur + id else cur - id
            p[Keys.STARTED] = true
        }
    }

    /** «Начать всё сначала»: сбрасывает отметки и снимает блокировку режима. */
    suspend fun reset() {
        context.dataStore.edit {
            it[Keys.STARTED] = false
            it[Keys.DONE] = emptySet()
        }
    }
}

package ru.example.prichastie

import androidx.compose.foundation.clickable
import androidx.compose.foundation.layout.Row
import androidx.compose.foundation.layout.fillMaxWidth
import androidx.compose.foundation.layout.padding
import androidx.compose.foundation.lazy.LazyColumn
import androidx.compose.foundation.lazy.items
import androidx.compose.material3.Checkbox
import androidx.compose.material3.LinearProgressIndicator
import androidx.compose.material3.MaterialTheme
import androidx.compose.material3.OutlinedButton
import androidx.compose.material3.Text
import androidx.compose.runtime.Composable
import androidx.compose.runtime.getValue
import androidx.compose.ui.Alignment
import androidx.compose.ui.Modifier
import androidx.compose.ui.unit.dp
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import androidx.lifecycle.viewmodel.compose.viewModel
import ru.example.prichastie.model.Catalog

@Composable
fun StepListScreen(
    modifier: Modifier = Modifier,
    vm: MainViewModel = viewModel()
) {
    val ui by vm.ui.collectAsStateWithLifecycle()
    val visible = ui.steps.filter { it.included }
    val doneCount = visible.count { it.done }

    LazyColumn(modifier = modifier.padding(horizontal = 16.dp)) {
        item {
            Text("Весь чин", style = MaterialTheme.typography.headlineSmall)
            Text("$doneCount из ${visible.size} шагов")
            LinearProgressIndicator(
                progress = { ui.progress },
                modifier = Modifier
                    .fillMaxWidth()
                    .padding(vertical = 8.dp)
            )
            Row(
                verticalAlignment = Alignment.CenterVertically,
                modifier = Modifier
                    .fillMaxWidth()
                    .clickable(enabled = !ui.modeLocked) { vm.setFullMode(!ui.fullMode) }
            ) {
                Checkbox(
                    checked = ui.fullMode,
                    onCheckedChange = null,
                    enabled = !ui.modeLocked
                )
                Text(
                    "Полное прохождение (с тремя канонами)",
                    modifier = Modifier.padding(start = 8.dp)
                )
            }
        }

        Catalog.groups.forEach { group ->
            val rows = visible.filter { it.step.groupId == group.id }
            if (rows.isNotEmpty()) {
                item(key = "group_${group.id}") {
                    Text(
                        group.title,
                        style = MaterialTheme.typography.titleMedium,
                        modifier = Modifier.padding(top = 16.dp)
                    )
                    Text(group.subtitle, style = MaterialTheme.typography.bodySmall)
                }
                items(rows, key = { it.step.id }) { row ->
                    Row(
                        verticalAlignment = Alignment.CenterVertically,
                        modifier = Modifier
                            .fillMaxWidth()
                            .clickable { vm.setDone(row.step.id, !row.done) }
                    ) {
                        Checkbox(checked = row.done, onCheckedChange = null)
                        Text(
                            "${row.number}. ${row.step.title}",
                            modifier = Modifier.padding(start = 8.dp)
                        )
                    }
                }
            }
        }

        item {
            OutlinedButton(
                onClick = { vm.restart() },
                modifier = Modifier.padding(vertical = 16.dp)
            ) { Text("Начать всё сначала") }
        }
    }
}

package ru.example.prichastie.model

import org.junit.Assert.assertEquals
import org.junit.Assert.assertTrue
import org.junit.Test

class CatalogTest {

    @Test
    fun idsAreUnique() {
        val ids = Catalog.steps.map { it.id }
        assertEquals(ids.size, ids.toSet().size)
    }

    @Test
    fun everyStepHasKnownGroup() {
        val groupIds = Catalog.groups.map { it.id }.toSet()
        assertTrue(Catalog.steps.all { it.groupId in groupIds })
    }

    @Test
    fun counts() {
        assertEquals(40, Catalog.steps.size)
        assertEquals(37, Catalog.activeSteps(includeCanons = false).size)
    }

    @Test
    fun creedSharesOneText() {
        val keys = Catalog.steps.filter { it.title == "Символ веры" }.map { it.textKey }
        assertEquals(listOf("creed", "creed"), keys)
    }
}
